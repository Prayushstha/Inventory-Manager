import { useState, useMemo } from "react";
import {
  ResponsiveContainer,
  BarChart,
  Bar,
  XAxis,
  YAxis,
  Tooltip,
  CartesianGrid,
} from "recharts";

/* ────────────────────────────────────────────────────────────────
   Design tokens — lifted straight from the existing dashboard shell
   (dark surface, orange brand accent, success/info/warning role colors)
   ──────────────────────────────────────────────────────────────── */
const T = {
  bg: "#121212",
  card: "#1a1a1a",
  cardBorder: "#2a2a28",
  divider: "#242422",
  textPrimary: "#f5f5f3",
  textSecondary: "#8f8f8a",
  textMuted: "#63635f",
  primary: "rgb(251, 129, 53)",
  primaryDim: "rgba(251, 129, 53, 0.14)",
  success: "#1D9E75",
  warning: "#e6a817",
  danger: "#e24b4a",
  info: "#378add",
};

function Eyebrow({ children, color = T.primary, tint = T.primaryDim }) {
  return (
    <span
      style={{
        display: "inline-block",
        fontSize: 11,
        fontWeight: 700,
        letterSpacing: "0.06em",
        color,
        background: tint,
        border: `1px solid ${color}`,
        borderRadius: 4,
        padding: "3px 8px",
        textTransform: "uppercase",
      }}
    >
      {children}
    </span>
  );
}

function Card({ accent, children, style }) {
  return (
    <div
      style={{
        background: T.card,
        border: `1px solid ${T.cardBorder}`,
        borderTop: accent ? `3px solid ${accent}` : `1px solid ${T.cardBorder}`,
        borderRadius: 10,
        padding: "20px 22px",
        ...style,
      }}
    >
      {children}
    </div>
  );
}

/* ────────────────────────────────────────────────────────────────
   1. Sales & revenue chart
   ──────────────────────────────────────────────────────────────── */
const MONTHLY_DATA = [
  { label: "1", revenue: 40 }, { label: "4", revenue: 90 }, { label: "7", revenue: 60 },
  { label: "10", revenue: 130 }, { label: "13", revenue: 80 }, { label: "16", revenue: 150 },
  { label: "19", revenue: 100 }, { label: "22", revenue: 170 }, { label: "25", revenue: 120 },
  { label: "28", revenue: 400 },
];
const YEARLY_DATA = [
  { label: "Jan", revenue: 1200 }, { label: "Feb", revenue: 1800 }, { label: "Mar", revenue: 900 },
  { label: "Apr", revenue: 2100 }, { label: "May", revenue: 1600 }, { label: "Jun", revenue: 2400 },
  { label: "Jul", revenue: 1300 }, { label: "Aug", revenue: 400 }, { label: "Sep", revenue: 0 },
  { label: "Oct", revenue: 0 }, { label: "Nov", revenue: 0 }, { label: "Dec", revenue: 0 },
];

function ChartTooltip({ active, payload, label }) {
  if (!active || !payload?.length) return null;
  return (
    <div
      style={{
        background: "#0f0f0f",
        border: `1px solid ${T.cardBorder}`,
        borderRadius: 6,
        padding: "8px 12px",
        fontSize: 12,
        color: T.textPrimary,
      }}
    >
      <div style={{ color: T.textSecondary, marginBottom: 2 }}>{label}</div>
      <div style={{ fontWeight: 700 }}>NPR {payload[0].value.toLocaleString()}</div>
    </div>
  );
}

export function SalesChart({ overViewTime }) {
  const data = overViewTime ? YEARLY_DATA : MONTHLY_DATA;
  const total = useMemo(() => data.reduce((s, d) => s + d.revenue, 0), [data]);

  return (
    <Card>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "flex-start", marginBottom: 18 }}>
        <div>
          <Eyebrow>trends</Eyebrow>
          <h3 style={{ margin: "10px 0 2px", fontSize: 17, fontWeight: 700, color: T.textPrimary }}>
            Sales &amp; revenue
          </h3>
          <p style={{ margin: 0, fontSize: 13, color: T.textMuted }}>
            {overViewTime ? "By month, this year" : "By day, this month"}
          </p>
        </div>
        <div style={{ textAlign: "right" }}>
          <div style={{ fontSize: 12, color: T.textMuted, textTransform: "uppercase", letterSpacing: "0.05em" }}>
            Total
          </div>
          <div style={{ fontSize: 20, fontWeight: 700, color: T.textPrimary }}>
            NPR {total.toLocaleString()}
          </div>
        </div>
      </div>

      <div style={{ width: "100%", height: 220 }}>
        <ResponsiveContainer>
          <BarChart data={data} margin={{ top: 4, right: 4, left: 4, bottom: 0 }}>
            <CartesianGrid vertical={false} stroke={T.divider} />
            <XAxis
              dataKey="label"
              axisLine={false}
              tickLine={false}
              tick={{ fill: T.textMuted, fontSize: 11 }}
            />
            <YAxis
              axisLine={false}
              tickLine={false}
              tick={{ fill: T.textMuted, fontSize: 11 }}
              width={36}
            />
            <Tooltip cursor={{ fill: "rgba(255,255,255,0.04)" }} content={<ChartTooltip />} />
            <Bar dataKey="revenue" fill={T.primary} radius={[3, 3, 0, 0]} maxBarSize={28} />
          </BarChart>
        </ResponsiveContainer>
      </div>
    </Card>
  );
}

function initials(name) {

}

  const max = Math.max(...CUSTOMERS.map((c) => c.spend));
  return (

      <Eyebrow color={T.warning} tint="rgba(230,168,23,0.14)">loyalty</Eyebrow>
      <h3 style={{ margin: "10px 0 16px", fontSize: 17, fontWeight: 700, color: T.textPrimary }}>
        Top customers
      </h3>

      <div style={{ display: "flex", flexDirection: "column", gap: 14 }}>
        {CUSTOMERS.map((c) => (
          <div key={c.name} style={{ display: "flex", alignItems: "center", gap: 12 }}>
            <div
              style={{
                width: 34,
                height: 34,
                borderRadius: "50%",
                background: T.primaryDim,
                color: T.primary,
                display: "flex",
                alignItems: "center",
                justifyContent: "center",

              </div>

}
